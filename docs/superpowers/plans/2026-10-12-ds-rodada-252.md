```ts
/**
 * A regra de esconder ao rolar das barras (`TopBar`, `BottomBar`,
 * `Toolbar sticky="hide"`): some ao descer mais de 4px, volta ao subir mais de
 * 4px, volta sempre no topo (até `mostrarAte`) e nunca some com o foco dentro.
 * Quem decide que só vale no telefone é o CSS de cada barra.
 */
export function proximoEscondido(
  atual: boolean,
  { y, antes, focada, mostrarAte = 0 }: { y: number; antes: number; focada: boolean; mostrarAte?: number }
): boolean {
  if (focada || y <= mostrarAte) return false
  if (y - antes > 4) return true
  if (antes - y > 4) return false
  return atual
}

/**
 * Quanto rolou quem disparou o `scroll`, se ele conta para `el`: o documento
 * (a janela dele, que na moldura do catálogo é a do iframe) ou um ancestral.
 * Decidido a cada evento — resolver o ancestral uma vez falhou na moldura,
 * onde o CSS chega depois do efeito.
 */
export function alvoDaRolagem(e: Event, el: Element): number | null {
  const doc = el.ownerDocument
  const alvo = e.target as Element | Document | null
  if (!alvo || alvo === doc) return (doc.defaultView ?? window).scrollY
  if ("contains" in alvo && alvo.contains(el)) return (alvo as Element).scrollTop
  return null
}
```

```ts
/**
 * Quem sai para o "⋯" quando a fileira não cabe (`Tabs`, `ToolbarActions`,
 * `PageHeaderActions`). Tira da `ponta` — as abas do fim, as ações do começo —
 * pulando os `fixos` (a aba ativa, a última ação) e quem `podeSair` recusa,
 * até a soma caber em `cabe` com a `reserva` do botão ⋯.
 */
export function quemSai(
  larguras: number[],
  cabe: number,
  { ponta, fixos, podeSair = () => true, reserva, gap = 0 }: {
    ponta: "inicio" | "fim"
    fixos: Set<number>
    podeSair?: (i: number) => boolean
    reserva: number
    gap?: number
  }
): number[] {
  const n = larguras.length
  const soma = (idx: number[]) =>
    idx.reduce((s, i) => s + larguras[i], 0) + gap * Math.max(0, idx.length - 1)
  const todos = larguras.map((_, i) => i)
  if (soma(todos) <= cabe + 0.5) return []
  const ordem = ponta === "fim" ? [...todos].reverse() : todos
  const ficam = new Set(todos)
  for (const i of ordem) {
    if (soma([...ficam]) + reserva + gap <= cabe + 0.5) break
    if (fixos.has(i) || !podeSair(i)) continue
    ficam.delete(i)
  }
  return todos.filter((i) => !ficam.has(i) && i < n)
}

/** Tira o `hidden` que o mecanismo pôs — ao desligar o transbordo. */
export function limparExcesso(els: Iterable<HTMLElement>) {
  for (const el of els) el.removeAttribute("hidden")
}
```
